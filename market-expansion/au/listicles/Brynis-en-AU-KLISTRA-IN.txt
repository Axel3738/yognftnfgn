(function(){
  var TOTAL = 45 * 60; // countdown starts at 45 min
  var els = document.querySelectorAll('.lst-wrap [data-t]');
  function tick(){
    var h = Math.floor(TOTAL/3600), m = Math.floor((TOTAL%3600)/60), s = TOTAL%60;
    els.forEach(function(e){
      var t = e.getAttribute('data-t');
      e.textContent = ('0'+(t==='h'?h:t==='m'?m:s)).slice(-2);
    });
    if(TOTAL>0) TOTAL--;
  }
  tick(); setInterval(tick,1000);
})();
